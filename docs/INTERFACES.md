## 3. Smart contracts (Solidity ^0.8.24, OpenZeppelin 5, all UUPS proxies)

Roles live in `InstallationRegistry` (AccessControl hub). `ADMIN` = `DEFAULT_ADMIN_ROLE` (0x00).
Other roles: `keccak256("OPERATOR_ROLE")`, `keccak256("VERIFIER_ROLE")`, `keccak256("ATTESTOR_ROLE")`,
`keccak256("REGULATOR_ROLE")`. The other two contracts call `registry.hasRole(...)`.

Global pause lives in the registry (`PausableUpgradeable`); `attest`, `verifierSign`, `allocate`
and `claim` revert with OZ `EnforcedPause()` while `registry.paused()`.

Upgrades: every contract's `_authorizeUpgrade` requires `msg.sender == registry.timelock()`.
The timelock is an OZ `TimelockController` with `minDelay = 86400`. Direct upgrade by admin reverts
`NotTimelock()`.

```solidity
enum Sector { STEEL, CEMENT }                                   // 0,1
enum Route  { BF_BOF, EAF, DRI_EAF, CEMENT_DRY, CEMENT_WET }    // 0..4
enum Status { None, PendingVerifier, Attested, Rejected }       // 0..3

struct Installation {
    bytes32 id; bytes2 country; Sector sector; Route route;
    uint64 capacityTpy; address operator; bool exists;
}

interface IInstallationRegistry {
    event InstallationRequested(bytes32 indexed id, address indexed requester, bytes2 country, Sector sector, Route route, uint64 capacityTpy);
    event InstallationRegistered(bytes32 indexed id, address indexed operator, bytes2 country, Sector sector, Route route, uint64 capacityTpy);
    error UnknownInstallation(bytes32 id);
    error InstallationExists(bytes32 id);
    error NotTimelock();
    event AttestorPqKeySet(address indexed attestor, bytes32 pqKeyHash);

    function initialize(address admin, address timelock) external;
    function requestInstallation(bytes32 id, bytes2 country, Sector sector, Route route, uint64 capacityTpy) external; // anyone; event only (no storage)
    function registerInstallation(bytes32 id, bytes2 country, Sector sector, Route route, uint64 capacityTpy, address operator) external; // ADMIN; also grants OPERATOR_ROLE to operator
    function getInstallation(bytes32 id) external view returns (Installation memory); // reverts UnknownInstallation
    function operatorOf(bytes32 id) external view returns (address);                   // address(0) if unknown
    function installationIds() external view returns (bytes32[] memory);
    function timelock() external view returns (address);
    function setAttestorPqKey(address attestor, bytes32 pqKeyHash) external; // ADMIN; anchors keccak256(raw ML-DSA-44 public key)
    function attestorPqKey(address attestor) external view returns (bytes32); // 0 if none
    function pause() external;   // ADMIN
    function unpause() external; // ADMIN
    // + OZ AccessControl: hasRole, grantRole, revokeRole, getRoleAdmin; OZ Pausable: paused()
    // + role constants as public bytes32: OPERATOR_ROLE, VERIFIER_ROLE, ATTESTOR_ROLE, REGULATOR_ROLE
}

struct AttestInput {
    bytes32 installationId;
    uint32  period;            // YYYYMM, e.g. 202609
    bytes32 commitment;        // keccak256(JCS(attestation document)) — see §5
    uint32  seeDeclared;       // tCO2 per t product × 1e4 (round half up)
    uint64  verifiedTonnes;    // whole tonnes of product
    uint8   riskScore;         // 0..100
    bytes32 modelVersionHash;  // sha256 of the frozen model artefact; must be approved
    uint8   algId;             // 1 = ECDSA-secp256k1 (EIP-191 over commitment) + ML-DSA-44 (FIPS 204)
    bytes32 pqSigHash;         // keccak256(raw ML-DSA-44 signature bytes)
    bytes32 cidDigest;         // sha2-256 digest of the bundle bytes = digest inside CIDv1(raw, sha2-256)
    bytes   ecdsaSig;          // 65-byte EIP-191 signature over `commitment`; signer must hold ATTESTOR_ROLE
}

struct Batch {
    bytes32 installationId; uint32 period; uint32 seeDeclared; uint64 verifiedTonnes;
    uint8 riskScore; uint8 algId; Status status; uint32 attestedAt;
    bytes32 commitment; bytes32 modelVersionHash; bytes32 pqSigHash; bytes32 cidDigest;
    address attestor;   // recovered ECDSA signer
    address verifier;   // set by verifierSign
    bytes32 reasonHash; // set by verifierSign
}

interface IEmissionsAttestation {
    event Attested(uint256 indexed batchId, bytes32 indexed installationId, uint32 period, bytes32 commitment, uint8 riskScore, Status status, string cid);
    event VerifierDecision(uint256 indexed batchId, address indexed verifier, bool approved, bytes32 reasonHash);
    event ModelApproved(bytes32 indexed modelVersionHash, uint8 index);
    event RiskThresholdSet(uint8 threshold);
    event ModelRevoked(bytes32 indexed modelVersionHash);
    error DuplicatePeriod(bytes32 installationId, uint32 period, uint256 existingBatchId);
    error UnknownBatch(uint256 batchId);
    error NotPending(uint256 batchId);
    error InvalidRiskScore(uint8 riskScore);
    error InvalidAlgId(uint8 algId);
    error ModelNotApproved(bytes32 modelVersionHash);
    error BadAttestorSignature(address recovered);
    error MissingRole(bytes32 role, address account);
    error NotTimelock();

    function initialize(address registry) external;
    function attest(AttestInput calldata a, string calldata cid) external returns (uint256 batchId);
        // ATTESTOR_ROLE on msg.sender; installation must exist; algId==1; riskScore<=100; model approved;
        // ecrecover(toEthSignedMessageHash(commitment), ecdsaSig) must hold ATTESTOR_ROLE;
        // (installationId, period) unused unless the previous batch for it is Rejected → else DuplicatePeriod;
        // status = riskScore > riskThreshold ? PendingVerifier : Attested. batchIds start at 1.
        // `cid` is emitted in the event only (not stored); the client rebuilds it from cidDigest.
    function verifierSign(uint256 batchId, bool approve, bytes32 reasonHash) external; // VERIFIER_ROLE; PendingVerifier → Attested|Rejected
    function getBatch(uint256 batchId) external view returns (Batch memory);           // reverts UnknownBatch
    function batchOf(bytes32 installationId, uint32 period) external view returns (uint256); // 0 if none
    function batchCount() external view returns (uint256);
    function allocInfo(uint256 batchId) external view returns (Status status, uint64 verifiedTonnes, bytes32 installationId); // slim view for the allocator
    function riskThreshold() external view returns (uint8);       // default 60
    function setRiskThreshold(uint8 t) external;                  // ADMIN
    function approveModel(bytes32 modelVersionHash) external;     // ADMIN
    function revokeModel(bytes32 modelVersionHash) external;      // ADMIN; emits ModelRevoked; later attests revert ModelNotApproved
    function isModelApproved(bytes32 modelVersionHash) external view returns (bool);
    function registry() external view returns (address);
}

struct Allocation { uint256 batchId; address importer; uint32 cnCode; uint64 tonnes; bool claimed; }

interface IConsignmentAllocator {
    event Allocated(uint256 indexed batchId, bytes32 indexed consignmentId, address indexed importer, uint32 cnCode, uint64 tonnes, uint64 remaining);
    event Claimed(uint256 indexed batchId, bytes32 indexed consignmentId, address indexed importer);
    error DoubleClaimBlocked(uint256 batchId, uint64 requested, uint64 remaining);
    error ConsignmentReused(bytes32 consignmentId);
    error BatchNotAttested(uint256 batchId);
    error NotBatchOperator(uint256 batchId, address caller);
    error NotConsignee(bytes32 consignmentId, address caller);
    error AlreadyClaimed(bytes32 consignmentId);
    error UnknownConsignment(bytes32 consignmentId);
    error ZeroTonnes();
    error ZeroImporter();
    error NotTimelock();

    function initialize(address registry, address attestation) external;
    function allocate(uint256 batchId, bytes32 consignmentId, address importer, uint32 cnCode, uint64 tonnes) external;
        // caller must be registry.operatorOf(batch.installationId); batch status Attested;
        // consignmentId never used → else ConsignmentReused; allocated + tonnes <= verifiedTonnes → else DoubleClaimBlocked
    function claim(bytes32 consignmentId) external;               // only allocation.importer; once
    function getAllocation(bytes32 consignmentId) external view returns (Allocation memory); // reverts UnknownConsignment
    function allocatedTonnes(uint256 batchId) external view returns (uint64);
    function remainingTonnes(uint256 batchId) external view returns (uint64);
}
```

## 5. Attestation document, signatures, bundle

Attestation **document** (what gets committed and signed):
```json
{
  "schema": "embedproof.attestation.v1",
  "report": { …report… },
  "assessment": { …ScoreResult minus latencyMs… },
  "chainId": 31337,
  "attestationContract": "0x…",
  "salt": "0x<32 random bytes hex>",
  "issuedAt": "2026-10-18T12:00:00Z"
}
```
- `canonical = JCS(document)` (RFC 8785), UTF-8 bytes. Python: `rfc8785.dumps`. JS: npm `canonicalize`.
- `commitment = keccak256(canonical)`.
- ECDSA: `personal_sign` (EIP-191) over the 32 raw commitment bytes with the attestor key.
  Python `Account.sign_message(encode_defunct(primitive=commitment), key)`; viem
  `recoverMessageAddress({ message: { raw: commitment }, signature })`; Solidity
  `ECDSA.recover(MessageHashUtils.toEthSignedMessageHash(commitment), sig)`.
- ML-DSA-44: pure ML-DSA (FIPS 204), empty context, over the **canonical bytes**. Backend order:
  `oqs` (liboqs-python) → `pqcrypto` (PQClean C) → `dilithium_py` (labelled reference-only).
  Browser: `@noble/post-quantum` `ml_dsa44`.
- `pqSigHash = keccak256(mldsaSignature)`.

**Bundle** (the file pinned to IPFS; JSON, UTF-8):
```json
{
  "schema": "embedproof.bundle.v1",
  "algId": 1,
  "batchHint": { "chainId": 31337, "installationId": "TR-EAF-0001", "period": 202609 },
  "commitment": "0x…",
  "ecdsa": { "scheme": "secp256k1-eip191-keccak256", "signer": "0x…", "signature": "0x…" },
  "mldsa": { "scheme": "ML-DSA-44", "backend": "pqclean", "publicKey": "<base64>", "signature": "<base64>" },
  "encryption": { "alg": "AES-256-GCM", "iv": "<base64 12 bytes>", "aad": "commitment", "ciphertext": "<base64 ciphertext||tag>" }
}
```
- Plaintext = `canonical` bytes. AAD = the 32 raw commitment bytes. Key = 32 random bytes per bundle.
- CID = CIDv1, codec raw (0x55), multihash sha2-256 of the bundle bytes, base32 lower with `b`
  prefix (`bafkrei…`). `cidDigest` on-chain = that sha2-256 digest.
- Key distribution: the attestor stores the key; the certificate / QR link carries it in the URL
  fragment `…/verify/<batchId>#k=<base64url key>` (fragments never reach a server).

## 6. Attestor HTTP API (FastAPI, `http://127.0.0.1:8000`, CORS for the web origin)

| Method & path | Body | Returns |
| --- | --- | --- |
| `GET /health` | – | `{ ok, chainId, attestor, modelVersionHash, pqBackend, ipfsMode, pqPublicKeyHash }` (`pqPublicKeyHash` = keccak256 of the raw ML-DSA-44 public key) |
| `POST /score` | `Report` | `ScoreResult` (no auth; rate-limited) |
| `POST /attest` | `{ report: Report, signature: "0x…" }` — exporter's EIP-191 sig over raw `keccak256(JCS(report))`; signer must equal `registry.operatorOf(installationId)` | `AttestResult` |
| `GET /bundle/{cid}` | – | raw bundle bytes (`application/json`) — local IPFS fallback gateway |
| `GET /verify/{cid}` | – | `VerifyResult` using the stored key (FR-A7) |
| `POST /verify` | `{ cid, document?: object }` — optional (possibly tampered) document to check instead of the decrypted one | `VerifyResult` |
| `POST /disclosure/{batchId}` | `{ address, message, signature }` — message `EmbedProof disclosure\nbatch:<id>\nts:<unix>` (±300 s) | `{ keyB64url, document }` if signer holds VERIFIER or REGULATOR role, is the installation operator, or is an importer with an allocation on that batch; else 403 |
| `POST /reasons` | `{ batchId, text }` | `{ reasonHash }` (keccak256 of UTF-8 text; stored for display) |
| `GET /reasons/{reasonHash}` | – | `{ text }` |

`ScoreResult`:
```json
{ "riskScore": 0, "band": "green", "q05": 0.0, "q50": 0.0, "q95": 0.0,
  "expectedLow": 0.0, "expectedHigh": 0.0, "seeDeclared": 0.0,
  "cbamDefault": 0.0, "belowHalfDefault": false,
  "modelVersionHash": "0x…",
  "shap": [ { "feature": "route", "label": "Production route", "value": "EAF", "contribution": -0.41 } ],
  "explanation": "one-sentence human summary",
  "latencyMs": 0.0 }
```
Bands: green ≤ 30, amber 31–60, red > 60. `shap` = top 3 by |contribution| from the q50 model.

`AttestResult`: `{ batchId, txHash, status: "Attested"|"PendingVerifier", riskScore, band, cid, commitment, pqSigHash, ecdsaSignature, keyB64url, verifyPath: "/verify/<batchId>#k=<key>", score: ScoreResult, gasUsed }`

`VerifyResult`: `{ cid, batchId, commitmentOnChain, commitmentRecomputed, commitmentMatches, ecdsaValid, ecdsaSigner, mldsaValid, pqSigHashMatches, cidDigestMatches, fieldsMatch, pqKeyMatches, allValid }`
— `pqKeyMatches`: keccak256(bundle ML-DSA public key) == `registry.attestorPqKey(batch.attestor)` (non-zero). This
anchors the post-quantum key to the attestor's on-chain identity.
— `fieldsMatch`: the document's report/assessment agree with the on-chain batch (installationId, period → YYYYMM,
round(see×1e4) == seeDeclared, productionTonnes == verifiedTonnes, riskScore, modelVersionHash, chainId,
attestationContract). The ECDSA signature covers only the commitment, so this check is what binds the
on-chain numbers to the signed document. `allValid` = every boolean above is true.

Errors: JSON `{ "detail": "…" }` with 4xx; contract reverts are decoded to their custom-error name and args.

## 9. v1.1 security amendments (supersede anything above that conflicts)

From the adversarial review. Every component implements these.

**A. EIP-712 attestor signature (replaces "EIP-191 over commitment").** `AttestInput.ecdsaSig` is an
EIP-712 signature by the attestor key over:
- Domain: `{ name: "EmbedProof", version: "1", chainId: <chain id>, verifyingContract: <EmissionsAttestation proxy> }`
- Type: `Attestation(bytes32 installationId,uint32 period,bytes32 commitment,uint32 seeDeclared,uint64 verifiedTonnes,uint8 riskScore,bytes32 modelVersionHash,uint8 algId,bytes32 pqSigHash)`
- `cidDigest` is deliberately NOT in the struct (the bundle carries the signature, so it cannot sign its own digest).
- Contract: build the domain separator from constant name/version hashes + `block.chainid` + `address(this)` (no storage reads), recover with OZ `ECDSA.tryRecover`, signer must hold ATTESTOR_ROLE. View `hashAttestation(AttestInput calldata a) returns (bytes32)` returns the EIP-712 digest.
- Python: `eth_account.messages.encode_typed_data(full_message=…)` + `Account.sign_message`. viem: `recoverTypedDataAddress`.
- Bundle `ecdsa.scheme` = `"eip712-embedproof-attestation-v1"`.
- Verifiers rebuild the struct from the on-chain batch (and, with the key, the commitment recomputed from the document). Because period/installation/tonnes/score are signed, a replay to another period or with altered numbers fails; no extra commitment-uniqueness storage is needed.

**B. Operators.** `allocate` requires `registry.hasRole(OPERATOR_ROLE, msg.sender) && registry.operatorOf(id) == msg.sender`.
Registry: `setOperator(bytes32 id, address newOperator)` (ADMIN; emits `OperatorChanged(bytes32 indexed id, address indexed previous, address indexed next)`; grants OPERATOR_ROLE to `next`). The attestor's `/attest` exporter check also requires OPERATOR_ROLE.

**C. Batch revocation and live attestor check.** EmissionsAttestation: `revokeBatch(uint256 batchId, bytes32 reasonHash)` (ADMIN; PendingVerifier|Attested → Rejected; emits `BatchRevoked(uint256 indexed batchId, address indexed by, bytes32 reasonHash)`; reasonHash ≠ 0). `allocInfo` now returns `(Status status, uint64 verifiedTonnes, bytes32 installationId, address attestor)`; `allocate` reverts `AttestorRevoked(uint256 batchId, address attestor)` if the batch's attestor no longer holds ATTESTOR_ROLE. `setRiskThreshold` is callable only by the timelock (`NotTimelock`).

**D. Verifier separation of duties.** `verifierSign` reverts `VerifierConflict(uint256 batchId, address verifier)` if the verifier is the batch attestor or the installation operator, and `ZeroReasonHash()` if `reasonHash == 0`.

**E. PQ key history.** Registry keeps every anchored key: `isAttestorPqKey(address attestor, bytes32 pqKeyHash) returns (bool)` (true once anchored via `setAttestorPqKey`, until revoked); `revokeAttestorPqKey(address attestor, bytes32 pqKeyHash)` (ADMIN; emits `AttestorPqKeyRevoked(address indexed attestor, bytes32 pqKeyHash)`). `attestorPqKey(address)` stays = the current key. `pqKeyMatches` uses `isAttestorPqKey(batch.attestor, keccak256(pk))`, so rotation no longer breaks history.

**F. Deploy safety.** On any chain other than 31337, `Deploy.s.sol` reverts unless `ATTESTOR`, `VERIFIER` and `REGULATOR` env addresses are set (never the public Anvil defaults).

**G. Exact SEE rounding (all components).** `seeDeclared = round_half_up(decimal(seeTco2PerT) × 10^4)` computed on the shortest round-trip decimal string of the number (Python `Decimal(repr(x))`; JS `Number.prototype.toString` handled as a decimal string, incl. exponent form) — never binary-float `Math.round(x*1e4)`.

**H. Attestor hardening.** Body-size cap enforced while streaming (chunked bodies included); generic 502 bodies (details only in logs; revert data parsed only from `exc.data`); rate limits on `/disclosure`, `/verify`, `/bundle`, `GET /reasons` too (IPv6 keyed by /64); cache `importers_of`; reject integers beyond 2^53 before float conversion; reject periods later than the current UTC month; `fieldsMatch` uses strict types/formats (ints are ints, `period` must fullmatch `YYYY-MM`, `chainId` int, no bool coercion).

Known and accepted (documented, not fixed): consignment ids are global (FR-C6 literal) so another registered operator can grief an id; cnCode is not bound to the batch; admin can grant roles instantly (only upgrades and the risk threshold are timelocked).

## 10. Round-3 amendments (judge and spec audit)

**I. Allocation cap per installation-period.** `ConsignmentAllocator` keys the running total by
`keccak256(abi.encode(installationId, period))`, not by batch. A revoked batch's allocations keep
counting, so a resubmission for the same installation-period cannot reset the cap.
`remaining(batchId) = verifiedTonnes − periodClaimed`, floored at 0. External signatures are unchanged.

**J. Capacity check.** The attestor returns 422 when `productionTonnes > capacity (t/yr) ÷ 12 × 1.2`,
using the nameplate capacity read on-chain from the registry.

**K. Assessment fields (additive).** `hasHistory: bool`; `flags: [{code, message}]` with codes
`GRID_FACTOR_BELOW_EMBER` (declared factor < 0.8 × Ember country-year) and
`ELECTRICITY_OUTSIDE_ROUTE_NORM` (MWh/t outside 0.5×–2× Climate TRACE's route assumption; a coarse
band, not a regulatory norm); `inputs: {…, artefactSha256}`, the exact scorer inputs. Flags never
change the score.

**L. Rescore.** `python -m attestor.rescore "<certificate URL>"` re-runs the scorer on
`assessment.inputs`. It checks sha256(artefact) == on-chain `modelVersionHash`, the recomputed score ==
on-chain `riskScore`, and that the commitment recomputes.

**M. Demo runner (testnet and local only).** It is enabled by `DEMO_RUNNER=1` and runs only on
chains 31337 and 84532, using server-held demo keys (`DEMO_EXPORTER_KEY`, `DEMO_VERIFIER_KEY`) and
importer addresses (`DEMO_IMPORTER_A`, `DEMO_IMPORTER_B`).
`GET /demo/status → {enabled, chainId}`.
`POST /demo/run/{green|red|double-claim} → {scenario, chainId, batchId, certificatePath, steps:[{label, ok, expected, txHash, detail}]}`.
Errors: 404 when disabled, 409 while a run is in progress, 429 with `Retry-After`.
The over-allocation in `double-claim` is sent with a fixed gas limit, so it is mined with status 0
and has an explorer link.
